<script type="text/python">
from browser import document, timer, window, ajax
import random, time

# =============================
# 設定
# =============================
GAME_DURATION = 30
TARGET_SIZE = 50

score = 0
shots = 0
game_active = False
spawn_time = 0
reaction_times = []

# =============================
# 遊戲場地
# =============================
game_area = document["brython_div1"]
game_area.style.width = "700px"
game_area.style.height = "450px"
game_area.style.margin = "20px auto"
game_area.style.border = "2px solid #555"
game_area.style.backgroundColor = "#111"
game_area.style.position = "relative"
game_area.style.cursor = "crosshair"
game_area.style.overflow = "hidden"

# =============================
# HUD 顯示
# =============================
hud = document.createElement("div")
hud.style.position = "absolute"
hud.style.top = "10px"
hud.style.left = "10px"
hud.style.color = "#0f0"
hud.style.fontFamily = "monospace"
hud.style.fontSize = "1.1em"
game_area <= hud

timer_display = document.createElement("div")
timer_display.style.position = "absolute"
timer_display.style.top = "10px"
timer_display.style.right = "10px"
timer_display.style.color = "#0f0"
timer_display.style.fontFamily = "monospace"
timer_display.style.fontSize = "1.1em"
game_area <= timer_display

# =============================
# 目標圓點
# =============================
target = document.createElement("div")
target.style.width = f"{TARGET_SIZE}px"
target.style.height = f"{TARGET_SIZE}px"
target.style.borderRadius = "50%"
target.style.backgroundColor = "#00aaff" # 亮天藍
target.style.position = "absolute"
target.style.display = "none"
game_area <= target

# =============================
# 核心邏輯
# =============================
def spawn_target():
global spawn_time
max_x = game_area.clientWidth - TARGET_SIZE
max_y = game_area.clientHeight - TARGET_SIZE

x = random.randint(0, max_x)
y = random.randint(40, max_y)

target.style.left = f"{x}px"
target.style.top = f"{y}px"
target.style.display = "block"

spawn_time = time.time()

# --- 命中事件 ---
@target.bind("click")
def hit(event):
global score, shots

if not game_active:
return

score += 1
shots += 1

reaction = time.time() - spawn_time
reaction_times.append(reaction)

# 命中閃白效果
target.style.backgroundColor = "#ffffff"
def restore_color():
target.style.backgroundColor = "#00aaff"
timer.set_timeout(restore_color, 100)

target.style.display = "none"
spawn_target()

update_hud()
event.stopPropagation()

# --- Miss 點擊事件 ---
@game_area.bind("click")
def miss(event):
global shots
if not game_active:
return

shots += 1
# Miss 閃紅
original_color = game_area.style.backgroundColor
game_area.style.backgroundColor = "#440000"
def restore_bg():
game_area.style.backgroundColor = original_color
timer.set_timeout(restore_bg, 100)

update_hud()

def update_hud():
accuracy = (score / shots * 100) if shots else 0
avg_reaction = (sum(reaction_times) / len(reaction_times)) if reaction_times else 0
hud.text = (
f"Hits: {score} "
f"Shots: {shots} "
f"Accuracy: {accuracy:.1f}% "
f"Avg RT: {avg_reaction*1000:.0f}ms"
)

# =============================
# 計時
# =============================
def check_time():
global game_active
remain = int(end_time - time.time())

if remain <= 0:
game_active = False
target.style.display = "none"
timer_display.text = "TIME UP"
submit_score()
return

timer_display.text = f"{remain}s"
timer.set_timeout(check_time, 1000)

# =============================
# 上傳分數
# =============================
def submit_score():
data = {
"score": score,
"shots": shots,
"accuracy": score / shots if shots else 0
}

req = ajax.ajax()
req.open("POST", "{{ url_for('submit_score') }}", True)
req.set_header("Content-Type", "application/json")
req.send(window.JSON.stringify(data))

# =============================
# 開始遊戲
# =============================
def start_game():
global score, shots, reaction_times, end_time, game_active

score = 0
shots = 0
reaction_times = []
game_active = True

end_time = time.time() + GAME_DURATION
update_hud()
spawn_target()
check_time()

start_game()
</script>

🔹 改進說明

圓點顏色固定亮天藍：#00aaff

命中閃白：命中瞬間變白，0.1 秒後回亮天藍

Miss 閃紅：點擊空白背景時背景變暗紅，0.1 秒後恢復

HUD 顯示：

Hits（命中次數）

Shots（射擊次數）

Accuracy（命中率）

Avg RT（平均反應時間毫秒）

計時器：30 秒倒數

Ajax 上傳分數：遊戲結束自動提交
